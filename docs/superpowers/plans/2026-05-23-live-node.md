# Phase 3: Live Node Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a runnable `qdht-node` CLI that connects real processes over WebSockets, stores content on disk, and exposes a Unix socket RPC for `put`, `get`, `peers`, and `replicas` commands.

**Architecture:** Service-oriented — `PeerManager` owns WebSocket connections, `SyncManager` dispatches protocol messages and runs delta sync, `ContentStore` owns filesystem piece storage, and `QDHTNode` orchestrates them. The `bin/qdht-node.ts` CLI is the single entry point for both the daemon (`start`) and client-side commands (`put`, `get`, `peers`, `replicas`).

**Tech Stack:** Node.js ESM, TypeScript strict, `ws` (WebSocket server + client), `commander` (CLI), `vitest` (tests), `tsx` (run without compile), `nostr-tools` (signing/verification from existing core).

// src/node/config.ts
import { readFile, writeFile, mkdir } from 'node:fs/promises'
import { dirname } from 'node:path'
import { generateKeypair } from '../core/identity/keys.js'

export interface QDHTConfig {
  identity: { privkey: string }
  peers: string[]
  port: number
  dataDir: string
}

export interface ConfigOverrides {
  port?: number
  peers?: string[]
  dataDir?: string
}

function validateConfig(raw: unknown, path: string): QDHTConfig {
  if (!raw || typeof raw !== 'object') {
    throw new Error(`Config at ${path} is not an object`)
  }
  const r = raw as Record<string, unknown>

  if (!r['identity'] || typeof r['identity'] !== 'object') {
    throw new Error(`Config missing identity.privkey`)
  }
  const identity = r['identity'] as Record<string, unknown>
  if (typeof identity['privkey'] !== 'string') {
    throw new Error(`Config missing identity.privkey`)
  }
  if (!/^[0-9a-f]{64}$/i.test(identity['privkey'])) {
    throw new Error(`identity.privkey must be 64 hex chars`)
  }

  if (!Array.isArray(r['peers'])) {
    throw new Error(`Config missing peers array`)
  }
  if (typeof r['port'] !== 'number') {
    throw new Error(`Config missing port`)
  }
  if (typeof r['dataDir'] !== 'string') {
    throw new Error(`Config missing dataDir`)
  }

  return {
    identity: { privkey: identity['privkey'] },
    peers: r['peers'] as string[],
    port: r['port'],
    dataDir: r['dataDir'],
  }
}

export async function loadConfig(configPath: string, overrides: ConfigOverrides = {}): Promise<QDHTConfig> {
  const raw = JSON.parse(await readFile(configPath, 'utf8')) as unknown
  const cfg = validateConfig(raw, configPath)
  if (overrides.port !== undefined) cfg.port = overrides.port
  if (overrides.peers !== undefined) cfg.peers = overrides.peers
  if (overrides.dataDir !== undefined) cfg.dataDir = overrides.dataDir
  return cfg
}

export async function initConfig(configPath: string, defaultDataDir: string): Promise<QDHTConfig> {
  try {
    return await loadConfig(configPath)
  } catch (err: unknown) {
    if ((err as NodeJS.ErrnoException).code !== 'ENOENT') {
      throw err
    }
  }

  const kp = generateKeypair()
  const cfg: QDHTConfig = {
    identity: { privkey: kp.privkey },
    peers: [],
    port: 7777,
    dataDir: defaultDataDir,
  }
  await mkdir(dirname(configPath), { recursive: true })
  await writeFile(configPath, JSON.stringify(cfg, null, 2))
  return cfg
}
```

// src/node/content-store.ts
import { createHash } from 'node:crypto'
import { readFile, writeFile, mkdir, stat } from 'node:fs/promises'
import { join } from 'node:path'

const PIECE_SIZE = 512 * 1024 // 512 KB

export interface ContentLocation {
  qkey: string
  hash: string
  totalPieces: number
  pieceSize: number
  sizeBytes: number
  name?: string
  mime?: string
  createdAt: string
}

export interface PutMeta {
  name?: string
  mime?: string
  ttl: number
}

interface IndexEntry {
  hash: string
  totalPieces: number
  pieceSize: number
  sizeBytes: number
  name?: string
  mime?: string
  createdAt: string
}

type Index = Record<string, IndexEntry>

export class ContentStore {
  private indexPath: string

  constructor(private dataDir: string) {
    this.indexPath = join(dataDir, 'index.json')
  }

  private sha256(data: Buffer): string {
    return createHash('sha256').update(data).digest('hex')
  }

  private async readIndex(): Promise<Index> {
    try {
      const raw = await readFile(this.indexPath, 'utf8')
      return JSON.parse(raw) as Index
    } catch {
      return {}
    }
  }

  private async writeIndex(index: Index): Promise<void> {
    await mkdir(this.dataDir, { recursive: true })
    await writeFile(this.indexPath, JSON.stringify(index, null, 2))
  }

  async put(data: Buffer, meta: PutMeta): Promise<ContentLocation> {
    const hash = this.sha256(data)
    const pieceDir = join(this.dataDir, hash)
    await mkdir(pieceDir, { recursive: true })

    const pieces: Buffer[] = []
    for (let offset = 0; offset < data.length; offset += PIECE_SIZE) {
      pieces.push(data.subarray(offset, offset + PIECE_SIZE))
    }
    if (pieces.length === 0) pieces.push(Buffer.alloc(0))

    for (let i = 0; i < pieces.length; i++) {
      await writeFile(join(pieceDir, `${i}.bin`), pieces[i]!)
    }

    const entry: IndexEntry = {
      hash,
      totalPieces: pieces.length,
      pieceSize: PIECE_SIZE,
      sizeBytes: data.length,
      name: meta.name,
      mime: meta.mime,
      createdAt: new Date().toISOString(),
    }

    const index = await this.readIndex()
    index[hash] = entry
    await this.writeIndex(index)

    return {
      qkey: hash,
      hash,
      totalPieces: pieces.length,
      pieceSize: PIECE_SIZE,
      sizeBytes: data.length,
      name: meta.name,
      mime: meta.mime,
      createdAt: entry.createdAt,
    }
  }

  async hasPiece(hash: string, index: number): Promise<boolean> {
    try {
      await stat(join(this.dataDir, hash, `${index}.bin`))
      return true
    } catch {
      return false
    }
  }

  async getPiece(hash: string, index: number): Promise<Buffer | null> {
    try {
      return await readFile(join(this.dataDir, hash, `${index}.bin`))
    } catch {
      return null
    }
  }

  async getIndex(): Promise<Index> {
    return this.readIndex()
  }
}
```

// src/node/peer-manager.ts
import { WebSocketServer, WebSocket, type RawData } from 'ws'

export interface PeerInfo {
  pubkey: string
  url: string
  latencyMs: number | null
  connectedAt: string
}

interface ConnectedPeer {
  ws: WebSocket
  pubkey: string
  url: string
  connectedAt: string
  lastPong: number
  latencyMs: number | null
}

export interface PeerManagerOptions {
  port: number
  pubkey: string
  privkey: string
}

type MessageHandler = (msg: unknown, peerId: string) => void

const PING_INTERVAL_MS = 30_000
const PONG_TIMEOUT_MS = 90_000
const RECONNECT_MIN_MS = 1_000
const RECONNECT_MAX_MS = 60_000
const STABLE_RESET_MS = 30_000

export class PeerManager {
  private server: WebSocketServer | null = null
  private peers_: Map<string, ConnectedPeer> = new Map()
  private messageHandlers: MessageHandler[] = []
  private pingTimer: ReturnType<typeof setInterval> | null = null
  private reconnectTimers: Map<string, ReturnType<typeof setTimeout>> = new Map()
  private closed = false

  constructor(private opts: PeerManagerOptions) {}

  async listen(): Promise<void> {
    return new Promise((resolve) => {
      this.server = new WebSocketServer({ port: this.opts.port })
      this.server.on('connection', (ws, req) => {
        const url = `ws://${req.socket.remoteAddress}:${req.socket.remotePort}`
        this.setupPeer(ws, url)
      })
      this.server.on('listening', () => {
        this.startPingLoop()
        resolve()
      })
    })
  }

  connect(url: string): void {
    if (this.closed) return
    const ws = new WebSocket(url)
    ws.on('open', () => this.setupPeer(ws, url))
    ws.on('error', () => this.scheduleReconnect(url))
    ws.on('close', () => this.scheduleReconnect(url))
  }

  private scheduleReconnect(url: string, delayMs = RECONNECT_MIN_MS): void {
    if (this.closed) return
    if (this.reconnectTimers.has(url)) return
    const timer = setTimeout(() => {
      this.reconnectTimers.delete(url)
      if (!this.closed) this.connect(url)
    }, delayMs)
    this.reconnectTimers.set(url, timer)
  }

  private setupPeer(ws: WebSocket, url: string): void {
    const tempId = `${url}-${Date.now()}`
    const peer: ConnectedPeer = {
      ws,
      pubkey: '',
      url,
      connectedAt: new Date().toISOString(),
      lastPong: Date.now(),
      latencyMs: null,
    }

    // Send our handshake
    ws.send(JSON.stringify({ type: 'handshake', pubkey: this.opts.pubkey }))

    ws.on('message', (data: RawData) => {
      let parsed: unknown
      try {
        parsed = JSON.parse(data.toString())
      } catch {
        return
      }

      // Handle handshake
      if (parsed && typeof parsed === 'object' && (parsed as Record<string, unknown>)['type'] === 'handshake') {
        const pubkey = (parsed as Record<string, unknown>)['pubkey'] as string
        peer.pubkey = pubkey
        this.peers_.delete(tempId)
        this.peers_.set(pubkey, peer)
        return
      }

      const peerId = peer.pubkey || tempId
      if (!this.peers_.has(peerId)) {
        this.peers_.set(peerId, peer)
      }

      for (const handler of this.messageHandlers) {
        handler(parsed, peerId)
      }
    })

    ws.on('pong', () => {
      peer.lastPong = Date.now()
    })

    ws.on('close', () => {
      const id = peer.pubkey || tempId
      this.peers_.delete(id)
    })

    ws.on('error', () => {
      const id = peer.pubkey || tempId
      this.peers_.delete(id)
    })

    if (!peer.pubkey) {
      this.peers_.set(tempId, peer)
    }
  }

  private startPingLoop(): void {
    this.pingTimer = setInterval(() => {
      const now = Date.now()
      for (const [id, peer] of this.peers_) {
        if (now - peer.lastPong > PONG_TIMEOUT_MS) {
          peer.ws.terminate()
          this.peers_.delete(id)
          continue
        }
        if (peer.ws.readyState === WebSocket.OPEN) {
          peer.ws.ping()
        }
      }
    }, PING_INTERVAL_MS)
  }

  broadcast(msg: unknown, excludePeerId?: string): void {
    const data = JSON.stringify(msg)
    for (const [id, peer] of this.peers_) {
      if (id === excludePeerId) continue
      if (peer.ws.readyState === WebSocket.OPEN) {
        peer.ws.send(data)
      }
    }
  }

  send(peerId: string, msg: unknown): void {
    const peer = this.peers_.get(peerId)
    if (peer && peer.ws.readyState === WebSocket.OPEN) {
      peer.ws.send(JSON.stringify(msg))
    }
  }

  peers(): PeerInfo[] {
    return Array.from(this.peers_.values()).map((p) => ({
      pubkey: p.pubkey,
      url: p.url,
      latencyMs: p.latencyMs,
      connectedAt: p.connectedAt,
    }))
  }

  onMessage(handler: MessageHandler): void {
    this.messageHandlers.push(handler)
  }

  async close(): Promise<void> {
    this.closed = true
    if (this.pingTimer) clearInterval(this.pingTimer)
    for (const timer of this.reconnectTimers.values()) clearTimeout(timer)
    for (const peer of this.peers_.values()) peer.ws.terminate()
    return new Promise((resolve) => {
      if (this.server) {
        this.server.close(() => resolve())
      } else {
        resolve()
      }
    })
  }
}

// src/node/sync-manager.ts
import { NeighbourStateMap } from '../core/neighbour-state.js'
import { Propagator } from '../core/propagation/propagator.js'
import { buildAnnouncement, isValidAnnouncement, getTag } from '../core/protocol/announcement.js'
import { buildDeltaRequest, buildDeltaResponse } from '../core/protocol/delta.js'
import { signAnnouncement } from '../core/identity/signing.js'

export interface SyncManagerOptions {
  pubkey: string
  privkey: string
  propagator: Propagator
  neighbourState: NeighbourStateMap
  broadcast: (msg: unknown, excludePeerId?: string) => void
  send: (peerId: string, msg: unknown) => void
}

interface AnyEvent {
  kind: number
  pubkey: string
  created_at: number
  tags: string[][]
  content: string
  sig: string
  id?: string
}

export class SyncManager {
  // Events indexed by id for delta responses
  private eventLog: Map<string, AnyEvent> = new Map()

  constructor(private opts: SyncManagerOptions) {}

  handleMessage(msg: unknown, fromPeerId: string): void {
    if (!msg || typeof msg !== 'object') return
    const event = msg as AnyEvent

    switch (event.kind) {
      case 10800:
        this.handleAnnouncement(event, fromPeerId)
        break
      case 10801:
        this.handleReplica(event, fromPeerId)
        break
      case 20800:
        this.handleDeltaRequest(event, fromPeerId)
        break
      case 20801:
        this.handleDeltaResponse(event)
        break
    }
  }

  private handleAnnouncement(event: AnyEvent, fromPeerId: string): void {
    if (!isValidAnnouncement(event)) return
    const id = event.id ?? `${event.pubkey}-${event.created_at}`
    const qkey = getTag(event, 'qkey') ?? id
    const hash = getTag(event, 'hash') ?? ''

    this.opts.propagator.addNote(id, fromPeerId, event.pubkey, event.created_at)
    this.opts.neighbourState.recordInbound(qkey, hash, id, fromPeerId, event.created_at)
    this.eventLog.set(id, event)
    this.opts.broadcast(event, fromPeerId)
  }

  private handleReplica(event: AnyEvent, fromPeerId: string): void {
    const id = event.id ?? `${event.pubkey}-${event.created_at}`
    this.eventLog.set(id, event)
    // Replica state update handled externally via NeighbourStateMap.updateReplica
  }

  private handleDeltaRequest(event: AnyEvent, fromPeerId: string): void {
    const sinceTag = event.tags.find((t) => t[0] === 'since')
    const since = sinceTag ? Number(sinceTag[1]) : 0

    const announcements: string[] = []
    const replicas: string[] = []
    for (const [, ev] of this.eventLog) {
      if (ev.created_at >= since) {
        if (ev.kind === 10800) announcements.push(JSON.stringify(ev))
        else if (ev.kind === 10801) replicas.push(JSON.stringify(ev))
      }
    }

    const resp = buildDeltaResponse({
      pubkey: this.opts.pubkey,
      requestId: event.id ?? '',
      since,
      announcements,
      replicas,
      reputationDeltas: [],
      expired: [],
    })
    this.opts.send(fromPeerId, resp)
  }

  private handleDeltaResponse(event: AnyEvent): void {
    try {
      const content = JSON.parse(event.content) as {
        announcements: string[]
        replicas: string[]
        reputationDeltas: string[]
        expired: string[]
      }
      for (const raw of content.announcements) {
        const ann = JSON.parse(raw) as AnyEvent
        const id = ann.id ?? `${ann.pubkey}-${ann.created_at}`
        if (!this.eventLog.has(id)) {
          this.eventLog.set(id, ann)
        }
      }
    } catch {
      // malformed response — ignore
    }
  }

  sendDeltaRequest(peerId: string, since: number): void {
    const req = buildDeltaRequest({ pubkey: this.opts.pubkey, since })
    this.opts.send(peerId, req)
  }

  publishAnnouncement(opts: {
    qkey: string
    hash: string
    sizeBytes: number
    pieces: number
    pieceSize: number
    ttl: number
    mime?: string
    name?: string
  }): void {
    const ann = buildAnnouncement({ pubkey: this.opts.pubkey, ...opts })
    const signed = signAnnouncement(ann, this.opts.privkey)
    this.eventLog.set(signed.id, { ...signed })
    this.opts.broadcast(signed, undefined)
  }
}

// src/node/qdht-node.ts
import { createServer as createNetServer, type Server as NetServer } from 'node:net'
import { unlink } from 'node:fs/promises'
import { join } from 'node:path'
import { GraphState } from '../core/graph/graph-state.js'
import { Propagator } from '../core/propagation/propagator.js'
import { NeighbourStateMap } from '../core/neighbour-state.js'
import { keypairFromHex } from '../core/identity/keys.js'
import { PeerManager } from './peer-manager.js'
import { SyncManager } from './sync-manager.js'
import { ContentStore, type PutMeta, type ContentLocation } from './content-store.js'
import type { QDHTConfig } from './config.js'

export class QDHTNode {
  private peerManager: PeerManager
  private syncManager: SyncManager
  private contentStore: ContentStore
  private rpcServer: NetServer | null = null
  private graph: GraphState
  private propagator: Propagator
  private neighbourState: NeighbourStateMap
  private kp: { pubkey: string; privkey: string }

  constructor(private config: QDHTConfig) {
    this.kp = keypairFromHex(config.identity.privkey)
    this.graph = new GraphState()
    this.graph.addNode(this.kp.pubkey)
    this.propagator = new Propagator(this.graph, this.graph.getIndex(this.kp.pubkey), 0.5)
    this.neighbourState = new NeighbourStateMap()
    this.contentStore = new ContentStore(config.dataDir)

    this.peerManager = new PeerManager({
      port: config.port,
      pubkey: this.kp.pubkey,
      privkey: this.kp.privkey,
    })

    this.syncManager = new SyncManager({
      pubkey: this.kp.pubkey,
      privkey: this.kp.privkey,
      propagator: this.propagator,
      neighbourState: this.neighbourState,
      broadcast: (msg, exclude) => this.peerManager.broadcast(msg, exclude),
      send: (peerId, msg) => this.peerManager.send(peerId, msg),
    })

    this.peerManager.onMessage((msg, peerId) => {
      this.syncManager.handleMessage(msg, peerId)
    })
  }

  async start(): Promise<void> {
    await this.peerManager.listen()

    for (const url of this.config.peers) {
      this.peerManager.connect(url)
    }

    await this.startRpc()
  }

  async stop(): Promise<void> {
    this.rpcServer?.close()
    const sockPath = join(this.config.dataDir, 'qdht.sock')
    try { await unlink(sockPath) } catch { /* ignore */ }
    await this.peerManager.close()
  }

  async put(data: Buffer, meta: PutMeta): Promise<ContentLocation> {
    const loc = await this.contentStore.put(data, meta)
    this.syncManager.publishAnnouncement({
      qkey: loc.qkey,
      hash: loc.hash,
      sizeBytes: loc.sizeBytes,
      pieces: loc.totalPieces,
      pieceSize: loc.pieceSize,
      ttl: meta.ttl,
      mime: meta.mime,
      name: meta.name,
    })
    return loc
  }

  pubkey(): string {
    return this.kp.pubkey
  }

  private async startRpc(): Promise<void> {
    const sockPath = join(this.config.dataDir, 'qdht.sock')
    try { await unlink(sockPath) } catch { /* ignore */ }

    return new Promise((resolve) => {
      this.rpcServer = createNetServer((conn) => {
        let buf = ''
        conn.on('data', (data) => {
          buf += data.toString()
          const lines = buf.split('\n')
          buf = lines.pop() ?? ''
          for (const line of lines) {
            if (!line.trim()) continue
            this.handleRpc(line.trim(), (resp) => {
              conn.write(JSON.stringify(resp) + '\n')
            })
          }
        })
        conn.on('error', () => {})
      })
      this.rpcServer.listen(sockPath, () => resolve())
    })
  }

  private handleRpc(line: string, reply: (resp: unknown) => void): void {
    let req: unknown
    try {
      req = JSON.parse(line)
    } catch {
      reply({ error: 'invalid JSON' })
      return
    }

    const r = req as Record<string, unknown>
    switch (r['cmd']) {
      case 'peers':
        reply({ peers: this.peerManager.peers() })
        break
      case 'replicas': {
        const key = r['key'] as string
        const state = this.neighbourState.get(key)
        if (!state) {
          reply({ replicas: [] })
          break
        }
        const replicas = Array.from(state.inbound.entries()).map(([pubkey, n]) => ({
          pubkey,
          lastSeen: new Date(n.lastSeen * 1000).toISOString(),
          pieceCount: n.pieceRanges.reduce((s, [a, b]) => s + (b - a + 1), 0),
          totalPieces: 0,
        }))
        reply({ replicas })
        break
      }
      case 'get': {
        const key = r['key'] as string
        this.contentStore.getIndex().then((index) => {
          const entry = index[key]
          if (!entry) {
            reply({ error: 'not found' })
            return
          }
          const pieces: Array<{ index: number; data: string }> = []
          const fetches: Promise<void>[] = []
          for (let i = 0; i < entry.totalPieces; i++) {
            const idx = i
            fetches.push(
              this.contentStore.getPiece(entry.hash, idx).then((buf) => {
                if (buf) pieces.push({ index: idx, data: buf.toString('base64') })
              }),
            )
          }
          Promise.all(fetches).then(() => {
            pieces.sort((a, b) => a.index - b.index)
            reply({ pieces })
          }).catch(() => reply({ error: 'read error' }))
        }).catch(() => reply({ error: 'index error' }))
        break
      }
      default:
        reply({ error: `unknown cmd: ${String(r['cmd'])}` })
    }
  }
}

// ── RPC helper ─────────────────────────────────────────────────────────
async function rpcCall(sockPath: string, request: unknown): Promise<unknown> {
  return new Promise((resolve, reject) => {
    try {
      const conn = netConnect(sockPath)
      let buf = ''
      conn.on('connect', () => conn.write(JSON.stringify(request) + '\n'))
      conn.on('data', (d) => {
        buf += d.toString()
        if (buf.includes('\n')) {
          conn.destroy()
          try { resolve(JSON.parse(buf.trim())) }
          catch { reject(new Error('bad JSON from RPC: ' + buf)) }
        }
      })
      conn.on('error', (err) => {
        if ((err as NodeJS.ErrnoException).code === 'ENOENT') {
          reject(new Error("Error: no running qdht-node found. Run 'qdht-node start' first."))
        } else {
          reject(err)
        }
      })
      setTimeout(() => reject(new Error('RPC timeout after 30s')), 30_000)
    } catch (err) {
      reject(err)
    }
  })
}

program.parseAsync(process.argv).catch((err: unknown) => {
  console.error(err instanceof Error ? err.message : String(err))
  process.exit(1)
})

// src/node/integration.test.ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest'
import { mkdtemp, rm } from 'node:fs/promises'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { QDHTNode } from './qdht-node.js'
import type { QDHTConfig } from './config.js'

let dirA: string
let dirB: string
let nodeA: QDHTNode
let nodeB: QDHTNode

async function waitFor(fn: () => boolean, timeoutMs = 5000): Promise<void> {
  const start = Date.now()
  while (!fn()) {
    if (Date.now() - start > timeoutMs) throw new Error('timeout waiting for condition')
    await new Promise((r) => setTimeout(r, 50))
  }
}

beforeEach(async () => {
  dirA = await mkdtemp(join(tmpdir(), 'qdht-int-a-'))
  dirB = await mkdtemp(join(tmpdir(), 'qdht-int-b-'))
})

afterEach(async () => {
  await nodeA?.stop()
  await nodeB?.stop()
  await rm(dirA, { recursive: true, force: true })
  await rm(dirB, { recursive: true, force: true })
})

describe('two-node integration', () => {
  it('node B receives announcement published by node A', async () => {
    const cfgA: QDHTConfig = {
      identity: { privkey: 'a'.repeat(64) },
      peers: [],
      port: 19950,
      dataDir: dirA,
    }
    const cfgB: QDHTConfig = {
      identity: { privkey: 'b'.repeat(64) },
      peers: ['ws://localhost:19950'],
      port: 19951,
      dataDir: dirB,
    }

    nodeA = new QDHTNode(cfgA)
    await nodeA.start()

    nodeB = new QDHTNode(cfgB)
    await nodeB.start()

    // Wait for B to connect to A
    await waitFor(() => nodeA.peerCount() > 0)

    const loc = await nodeA.put(Buffer.from('hello from A'), { name: 'a.txt', ttl: 3600 })

    // Wait for B to receive the announcement via broadcast
    await waitFor(() => nodeB.hasReceivedKey(loc.qkey), 5000)
    expect(nodeB.hasReceivedKey(loc.qkey)).toBe(true)
  })
})
